# MEDC — suggested crontab entries for the host running the lab
# Install with:  sudo crontab -e   (or drop into /etc/cron.d/medc)

# Boot-time health check (give the bridge + containers a minute to settle)
@reboot sleep 60 && /usr/local/bin/check-medc-health.sh > /var/log/medc-health-boot.log 2>&1

# Daily LXC backup at 02:00
0 2 * * * /usr/local/bin/backup-lxc-containers.sh > /var/log/lxc-backup.log 2>&1

# Weekly LXC maintenance on Sunday at 03:00
0 3 * * 0 /usr/local/bin/maintain-lxc-containers.sh > /var/log/lxc-maintenance.log 2>&1
